async function performTransaction(actions) {
    const transaction = await sequelize.transaction();
    try {
        const results = await Promise.all(actions.map(action => action(transaction)));
        await transaction.commit();
        return results;
    } catch (error) {
        await transaction.rollback();
        throw error;
    }
}
